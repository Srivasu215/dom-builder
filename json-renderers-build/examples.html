table.js (25 Lines) 
import render from "json-renderers-build";

// 1. Data records
const products = [
  { item: "Roller Bearing 6205", cat: "Parts", qty: 40, price: 280 },
  { item: "Synthetic Oil 5W-40", cat: "Lubes", qty: 15, price: 650 },
  { item: "Copper Cable 2.5mm",  cat: "Wire",  qty: 80, price: 120 }
];

// 2. Build table specification
const spec = render({
  type: "table",
  columns: ["item", "cat", "qty", "price"],
  data: products
});

console.log("Compiled Table AST:", spec);

⚡ Live Visual Output

1 Roller Bearing 6205 Parts 40 ₹280
2 Synthetic Oil 5W-40 Lubes 15 ₹650
3 Copper Cable 2.5mm Wire 80 ₹120
